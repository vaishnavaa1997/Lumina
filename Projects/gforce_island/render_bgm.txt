@echo off
setlocal
if "%~1"=="" (
    echo ============================================================================
    echo USAGE:
    echo   render_bgm.bat ^<project_name^> [mix_mode]
    echo.
    echo EXAMPLES:
    echo   render_bgm.bat my_video           ^(Renders VO-Ducked BGM: 0.18 bed / 0.55 swell^)
    echo   render_bgm.bat my_video bgmFull   ^(Renders Unducked Full BGM: 0.42 bed / 0.55 swell^)
    echo ============================================================================
    exit /b 1
)

set "PROJECT=%~1"
set "MIXMODE=%~2"
if "%MIXMODE%"=="" set "MIXMODE=bgmOnly"

echo.
echo [DK-STUDIO] Rendering Continuous Master BGM for project: "%PROJECT%" (Mode: %MIXMODE%)...
node "%~dp0render_bgm.js" "%PROJECT%" "%MIXMODE%"

if %ERRORLEVEL% NEQ 0 (
    echo.
    echo [ERROR] BGM rendering failed! Make sure bgm_master.html exists in projects\%PROJECT%\
    exit /b %ERRORLEVEL%
)

echo.
echo [SUCCESS] Saved continuous music track to: projects\%PROJECT%\build\bgm_master.wav
endlocal
